class Solution {
    public int[] twoSum(int[] numbers, int target) {
        int i =0;
        int j= numbers.length-1;
        int sum;
        while(i<j){
           sum = numbers[i]+numbers[j];
           if (target<sum) j--;
           else if(target> sum) i++;
           else{
            return new int[]{i+1,j+1}; 
        }
        }
    return new int[]{};
}
}
