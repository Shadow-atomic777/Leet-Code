class Solution {
    public int[] twoSum(int[] numbers, int target) {
        int sum;
        int len  = numbers.length;
        for (int i=0;i<len;i++){
           
            
                for(int j=i+1; j<len;j++){
                    sum = numbers[i]+numbers[j];
                    if(sum==target){
                        int[] ans={i+1,j+1};
                        return ans;
                    }
                } 
        }
        return new int[]{};
    }
}
